### 💡 今日学到的知识点

#### 1. Vue3 vs Vue2 对比
| 特性 | Vue2 (Options API) | Vue3 (Composition API) |
|------|-------------------|------------------------|
| 组件写法 | data/methods/computed分开 | setup()统一编写 |
| 响应式 | this.xxx = value | ref() / reactive() |
| 生命周期 | created/mounted等 | onMounted/onCreated等 |
| 逻辑复用 | Mixin(易冲突) | Composable函数(安全) |
| TypeScript支持 | 一般 | 优秀 |

#### 2. ref vs reactive 使用场景
```javascript
<script setup>
import { ref, reactive } from 'vue'

// ref: 用于基本数据类型(String/Number/Boolean)
const count = ref(0)
const name = ref('张三')

// 修改时需要.value
count.value++
console.log(name.value)

// reactive: 用于对象/数组
const formState = reactive({
  username: '',
  password: '',
  remember: true
})

// 修改时直接访问
formState.username = 'admin'

// 选择建议：
// - 单个值用ref
// - 表单对象用reactive
// - 需要解构时用ref(toRefs)
</script>
```

#### 3. computed 和 watch
```javascript
<script setup>
import { ref, computed, watch, watchEffect } from 'vue'

const firstName = ref('张')
const lastName = ref('三')
const age = ref(25)

// computed: 计算属性(有缓存)
const fullName = computed(() => `${firstName.value}${lastName.value}`)

// watch: 监听特定变量变化
watch(age, (newVal, oldVal) => {
  console.log(`年龄从${oldVal}变为${newVal}`)
})

// watchEffect: 自动追踪依赖(无需指定)
watchEffect(() => {
  console.log(`${firstName.value}今年${age.value}岁`)
})
</script>
```

### 🔐 路由守卫实现

```javascript
// 在router/index.js中添加
import { useAuthStore } from '@/store/auth'

router.beforeEach((to, from, next) => {
  const authStore = useAuthStore()

  // 设置页面标题
  document.title = to.meta.title ? `${to.meta.title} - 员工管理系统` : '员工管理系统'

  // 1. 不需要认证的页面直接放行(如登录页)
  if (to.meta.requiresAuth === false) {
    return next()
  }

  // 2. 未登录 → 跳转登录页
  if (!authStore.isLoggedIn) {
    return next({
      path: '/login',
      query: { redirect: to.fullPath }  // 记录原路径，登录后跳回
    })
  }

  // 3. 角色不匹配 → 回首页
  if (to.meta.roles && !to.meta.roles.includes(authStore.role)) {
    message.warning('没有权限访问此页面')
    return next('/dashboard')
  }

  next()
})
```


#### 3. 导航守卫类型
| 类型 | 触发时机 | 用途 |
|------|----------|------|
| `beforeEach` | 全局前置 | 登录校验、权限判断 |
| `afterEach` | 全局后置 | 设置页面标题、滚动位置 |
| `beforeEnter` | 路由独享 | 特定路由的额外校验 |
| `beforeRouteEnter` | 组件内 | 不能访问this(组件未创建) |
| `beforeUpdate` | 组件内 | 当前路由改变但组件复用 |
| `beforeLeave` | 组件内 | 离开确认(未保存提示) |

### 🗄️ Pinia Store - 认证状态 (store/auth.js)

```javascript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { authApi } from '@/api/auth'
import router from '@/router'
import { message } from 'ant-design-vue'

export const useAuthStore = defineStore('auth', {
  state: () => ({
    token: localStorage.getItem('token') || '',
    userId: localStorage.getItem('userId') || '',
    username: localStorage.getItem('username') || '',
    realName: localStorage.getItem('realName') || '',
    role: localStorage.getItem('role') || ''
  }),

  getters: {
    isLoggedIn: (state) => !!state.token,
    isAdmin: (state) => state.role === 'admin',
    userInfo: (state) => ({
      username: state.username,
      realName: state.realName,
      role: state.role
    })
  },

  actions: {
    async login({ username, password }) {
      try {
        const res = await authApi.login({ username, password })
        const data = res.data.data
        
        this.setUserInfo(data)
        
        message.success(`欢迎回来，${data.realName}！`)
        
        // 跳转到原页面或首页
        const redirect = router.currentRoute.value.query.redirect || '/'
        router.push(redirect)
        
        return res
      } catch (error) {
        throw error
      }
    },

    setUserInfo(data) {
      this.token = data.token
      this.userId = data.userId
      this.username = data.username
      this.realName = data.realName
      this.role = data.role

      // 持久化到localStorage
      localStorage.setItem('token', data.token)
      localStorage.setItem('userId', data.userId)
      localStorage.setItem('username', data.username)
      localStorage.setItem('realName', data.realName)
      localStorage.setItem('role', data.role)
    },

    logout() {
      this.token = ''
      this.userId = ''
      this.username = ''
      this.realName = ''
      this.role = ''

      localStorage.clear()
      
      router.push('/login')
    },

    updateProfile(info) {
      this.realName = info.realName
      localStorage.setItem('realName', info.realName)
    }
  }
})
```

### 🌐 Axios封装 (api/auth.js)

```javascript
import axios from 'axios'
import { useAuthStore } from '@/store/auth'
import router from '@/router'
import { message } from 'ant-design-vue'

// 创建Axios实例
const api = axios.create({
  baseURL: '/api',           // Vite代理会转发到后端
  timeout: 15000,            // 请求超时时间15秒
  headers: {
    'Content-Type': 'application/json'
  }
})

// 请求拦截器 - 自动携带Token
api.interceptors.request.use(
  config => {
    const authStore = useAuthStore()
    
    if (authStore.token) {
      config.headers.Authorization = `Bearer ${authStore.token}`
    }

    // 显示loading(可选)
    // loadingInstance = Loading.service()

    return config
  },
  error => {
    return Promise.reject(error)
  }
)

// 响应拦截器 - 统一处理响应和错误
api.interceptors.response.use(
  response => {
    const res = response.data
    
    // 根据后端约定的code判断业务状态
    if (res.code === 200) {
      return res  // 直接返回业务数据
    } else if (res.code === 401) {
      handleUnauthorized()
      return Promise.reject(new Error(res.message))
    } else if (res.code === 403) {
      message.error('没有权限执行此操作')
      return Promise.reject(new Error(res.message))
    } else {
      message.error(res.message || '操作失败')
      return Promise.reject(new Error(res.message))
    }
  },
  error => {
    // HTTP错误状态码处理
    if (error.response) {
      switch (error.response.status) {
        case 401:
          handleUnauthorized()
          break
        case 403:
          message.error('没有权限访问')
          break
        case 404:
          message.error('请求的资源不存在')
          break
        case 500:
          message.error('服务器内部错误')
          break
        default:
          message.error(error.response.data?.message || '网络异常')
      }
    } else if (error.message.includes('timeout')) {
      message.error('请求超时，请检查网络')
    } else if (error.message.includes('Network')) {
      message.error('网络连接失败，请检查网络')
    } else {
      message.error('未知错误')
    }

    return Promise.reject(error)
  }
)

  const authStore = useAuthStore()
  
  message.warning('登录已过期，请重新登录')
  authStore.logout()
}
```

> 5. **模块导出错误** → 统一API导入方式为对象调用"
